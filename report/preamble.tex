% ---------- Figure / screenshot helpers ----------
% Uses the real image if present at report/figures/<name>.png, otherwise draws a
% clearly labelled placeholder so the document always compiles. Replace the PNG
% later (same filename) and recompile -- no LaTeX changes needed.
\newcommand{\AppFigure}[3]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{figures/#1.png}{%
      \includegraphics[width=#2\linewidth]{figures/#1.png}%
    }{%
      \fbox{\begin{minipage}[c][0.42\linewidth][c]{#2\linewidth}
        \centering\color{ceMuted}\small
        \faCamera\ \textit{Screenshot placeholder}\\[4pt]
        \texttt{figures/#1.png}\\[4pt]
        {\itshape #3}
      \end{minipage}}%
    }
    \caption{#3}
    \label{fig:#1}
  \end{figure}
}

\newcommand{\Diagram}[3]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{diagrams/#1.png}{%
      \includegraphics[width=#2\linewidth]{diagrams/#1.png}%
    }{%
      \fbox{\begin{minipage}[c][0.3\linewidth][c]{#2\linewidth}
        \centering\color{ceCoral}\small Diagram not yet rendered: \texttt{#1.png}
      \end{minipage}}%
    }
    \caption{#3}
    \label{fig:#1}
  \end{figure}
}
